// ❌ Violates LSP: Square inherits Rectangle
class Rectangle {
    int width, height;
    void setWidth(int w){ width=w; }
    void setHeight(int h){ height=h; }
    int area(){ return width*height; }
}

class Square extends Rectangle {
    void setWidth(int w){ width=height=w; }
    void setHeight(int h){ width=height=h; }
}

// ✅ Correct approach
interface Shape {
    int area();
}

class Rectangle implements Shape { /* same as above */ }
class Square implements Shape {
    int side;
    public int area(){ return side*side; }
}


public class Green {
   public void getColor() {
      System.out.println("Green");
   }
}

public class Blue extends Green {
   public void getColor() {
      System.out.println("Blue");
   }
}

public class Main{
   public static void main(String[] args) {
      // violate LSP because color of green object is blue
      Green green = new Blue();
      green.getColor();
      //output: Blue
   }
}


// Solution :

public interface IColor{
   public void getColor();
}

public class Green implements IColor {
   public void getColor() {
      System.out.println("Green");
   }
}

public class Blue implements IColor {
   public void getColor() {
      System.out.println("Blue");
   }
}

public class Main{
   public static void main(String[] args) {
      IColor color = new Blue();
      color.getColor();
      //output: Blue
   }
}
